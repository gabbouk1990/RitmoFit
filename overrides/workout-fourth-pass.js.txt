function rf4Kind(t){
 const title=String((t?.kind||'')+' '+(t?.name||'')).toLowerCase();
 return /nuoto|piscina|swim/.test(title)?'nuoto':/hybrid|ibrido|functional|conditioning/.test(title)?'hybrid':'forza';
}

function rf4DateWithin(session,range){
 const date=String(session?.date||'').slice(0,10),today=inputDate();
 if(!/^\d{4}-\d{2}-\d{2}$/.test(date))return false;
 if(range==='year')return date.slice(0,4)===today.slice(0,4);
 if(range==='month')return date.slice(0,7)===today.slice(0,7);
 return date>=proIso(proWeekMonday(today))&&date<=today;
}
function rf4FilteredSessions(){return (state.sessions||[]).filter(x=>rf4DateWithin(x,rf4StatsRange));}
function rf4Stats(){
 const filtered=rf4FilteredSessions();
 const periods={week:'Settimana',month:'Mese',year:'Anno'};
 const totalMins=filtered.reduce((a,x)=>a+(Number(x.duration)||0),0);
 const totalSets=filtered.reduce((a,x)=>a+((x.completedSets||[]).reduce((b,v)=>b+(Number(v)||0),0)),0);
 const totalDays=new Set(filtered.map(s=>s.date)).size;
 const tab='<div class="rf4-period-tabs">'+Object.entries(periods).map(([k,v])=>
  '<button class="'+(rf4StatsRange===k?'active':'')+'" data-act="rf4StatsRange" data-range="'+k+'">'+v+'</button>').join('')+'</div>';
 // Data labels reflect the selected timeframe. No zero-filling absent health measurements.
 let buckets=[];
 const now=new Date(inputDate()+'T12:00:00');
 if(rf4StatsRange==='week'){
  const mon=proWeekMonday(inputDate());
  buckets=Array.from({length:7},(_,i)=>{const d=new Date(mon);d.setDate(mon.getDate()+i);return {key:proIso(d),label:['L','M','M','G','V','S','D'][i]};});
 }else if(rf4StatsRange==='month'){
  const count=new Date(now.getFullYear(),now.getMonth()+1,0).getDate();
  const spans=[[1,7],[8,14],[15,21],[22,count]];
  buckets=spans.map(([start,end],i)=>({key:i,label:'S'+(i+1),has:s=>{const day=Number(s.date.slice(8));return day>=start&&day<=end;}}));
 }else{
  buckets=Array.from({length:12},(_,i)=>({key:i,label:['G','F','M','A','M','G','L','A','S','O','N','D'][i],has:s=>Number(s.date.slice(5,7))===i+1}));
 }
 const data=buckets.map(b=>{
  const arr=filtered.filter(s=>b.has?b.has(s):s.date===b.key);
  return {...b,count:arr.length,mins:arr.reduce((a,s)=>a+(Number(s.duration)||0),0)};
 });
 const max=Math.max(1,...data.map(d=>d.mins));
 const chart=data.map(d=>'<div class="rf4-stat-bar"><strong>'+d.mins+'</strong><div><span style="height:'+Math.max(2,100*d.mins/max)+'%"></span></div><small>'+d.label+'</small></div>').join('');
 const recent=filtered.slice().sort((a,b)=>String(b.date).localeCompare(String(a.date))).slice(0,5);
 const recentRows=recent.map(s=>'<div class="rf4-recent-row"><span>✓</span><div><strong>'+rf4Safe(s.type||s.name||'Allenamento')+'</strong><small>'+rf4Safe(s.date)+' · '+fmt(s.duration)+' min</small></div><em>'+rf4Safe(s.rpe?'RPE '+fmt(s.rpe,1):'Registrata')+'</em></div>').join('');
 const details=rf4ExistingStats();
 return '<div class="rf4-app rf4-stats"><div class="rf4-screen-head"><div class="rf4-kicker">IL TUO ANDAMENTO</div><h2>Statistiche</h2><p>Progressi delle sedute registrate in Ritmo Fit.</p></div>'+tab+
 '<div class="rf4-stat-grid"><div><small>Sedute</small><strong>'+filtered.length+'</strong><span>nel periodo</span></div><div><small>Tempo</small><strong>'+Math.round(totalMins/60*10)/10+' h</strong><span>'+totalMins+' min totali</span></div><div><small>Serie</small><strong>'+totalSets+'</strong><span>registrate</span></div><div><small>Giorni</small><strong>'+totalDays+'</strong><span>attivi</span></div></div>'+
 '<section class="rf4-stats-card"><div class="rf4-section-heading"><h3>Tempo di allenamento</h3><span>minuti</span></div><div class="rf4-stats-bars">'+chart+'</div></section>'+
 '<section class="rf4-stats-card"><h3>Ultime sessioni</h3>'+(recentRows||'<p class="rf4-hint">Nessuna sessione manuale registrata per questo periodo.</p>')+'</section>'+
 '<details class="rf4-stats-details"><summary>Altre statistiche: carichi e volume <span>⌄</span></summary>'+details+'</details>'+
 '<p class="rf4-hint">Gli allenamenti importati da Apple Salute restano consultabili nello Storico. Qui non li sommo alle sessioni manuali per evitare duplicazioni.</p></div>';
}

// Keep all existing logging/editing handlers. These are navigation-only additions.
const rf4OldHandle=handleWorkoutProAction;
handleWorkoutProAction=async function(act,t){
 const id=t.dataset.id;
 switch(act){
  case 'rf4Category':
   rf4Category=t.dataset.category||'forza';rf4Stage='programs';render();return true;
  case 'rf4OpenTemplate':
   rf4TemplateId=id;rf4Stage='template';render();return true;
  case 'rf4OpenExercise':{
   const template=rf4Program(rf4TemplateId),idx=Number(t.dataset.index);
   if(!template||!Number.isInteger(idx)||idx<0||idx>=template.exercises.length)return true;
   rf4ExerciseIndex=idx;rf4ExerciseTab='esegui';rf4Stage='exercise';render();return true;
  }
  case 'rf4Back':
   rf4Stage=rf4Stage==='exercise'?'template':rf4Stage==='template'?'programs':'activities';render();return true;
  case 'rf4ExerciseTab':
   rf4ExerciseTab=t.dataset.tab||'esegui';render();return true;
  case 'rf4ExercisePrev':rf4ExerciseIndex=Math.max(0,rf4ExerciseIndex-1);rf4ExerciseTab='esegui';render();return true;
  case 'rf4ExerciseNext':{
   const template=rf4Program(rf4TemplateId);
   rf4ExerciseIndex=Math.min((template?.exercises?.length||1)-1,rf4ExerciseIndex+1);rf4ExerciseTab='esegui';render();return true;
  }
  case 'rf4Resume':state.workoutPane='timer';render();return true;
  case 'rf4TimerFocus':rf4TimerFocus=!rf4TimerFocus;render();return true;
  case 'rf4StatsRange':rf4StatsRange=t.dataset.range||'week';render();return true;
 }
 return rf4OldHandle(act,t);
};
