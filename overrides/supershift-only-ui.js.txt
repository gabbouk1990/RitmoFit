function sup2026Source(d){
 if(typeof d!=='string'||!/^2026-\d{2}-\d{2}$/.test(d))return null;
 return (state.shifts||[]).find(x=>x.date===d&&x.id==='secure-supershift-2026-'+d&&x.source==='ics')||null;
}
// Unica sorgente condivisa anche da Home, prossimi turni e allenamento suggerito.

const sup2026OriginalRecommended=recommended;
recommended=function(d){
 if(!sup2026Source(d))return {name:'Nessuna sessione proposta',why:'Giorno senza turno verificato nel calendario Supershift. Non viene applicata alcuna rotazione automatica.',key:null,icon:'📅'};
 return sup2026OriginalRecommended(d);
};
function sup2026MonthInfo(year,month){
 const max=new Date(year,month,0).getDate(),days=[];
 for(let i=1;i<=max;i++){const date=year+'-'+String(month).padStart(2,'0')+'-'+String(i).padStart(2,'0');const sh=sup2026Source(date);days.push({date,sh,info:SUP2026_LABELS[sh?.type]||null});}
 return days;
}
function sup2026Stat(label,value,description,icon){
 return '<div class="sup-kpi-card"><span class="sup-kpi-icon" aria-hidden="true">'+icon+'</span><div><small>'+label+'</small><strong>'+value+'</strong><em>'+description+'</em></div></div>';
}
function sup2026Legend(){
 return '<div class="sup-shift-legend">'+Object.values(SUP2026_LABELS).map(item=>
  '<div class="sup-legend-item"><span class="sup-tag '+item.tone+'">'+item.code+'</span><span>'+item.label+'</span></div>').join('')+'</div>';
}

// Navigation kept in existing delegated click handler (no new global listener).
const supPreviousWorkoutAction=handleWorkoutProAction;
handleWorkoutProAction=async function(act,t){
 if(act==='supDay'){
  if(!/^2026-\d{2}-\d{2}$/.test(t.dataset.date||''))return true;
  state.date=t.dataset.date;state.month=state.date.slice(0,7);render();return true;
 }
 if(act==='supMonth'){
  const n=Number(t.dataset.n)||0;
  const current=String(state.month||'2026-10').startsWith('2026-')?Number(state.month.slice(5)):10;
  const next=Math.max(1,Math.min(12,current+n));
  state.month='2026-'+String(next).padStart(2,'0');
  state.date=state.month+'-'+String(Math.min(Number(String(state.date||'').slice(8))||1,new Date(2026,next,0).getDate())).padStart(2,'0');
  render();return true;
 }
 if(act==='supToday'){
  const today=inputDate();
  state.date=today.startsWith('2026-')?today:'2026-10-17';
  state.month=state.date.slice(0,7);
  render();return true;
 }
 return supPreviousWorkoutAction(act,t);
};
