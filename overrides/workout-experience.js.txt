function proEsc(s){return safe(String(s??''));}
function proSetDetails(){

function proSuggestedReps(ex){
 return Math.max(1,Math.min(200,Number(String(ex?.reps||'').match(/\d+/)?.[0])||10));
}
function proSuggestedLoad(ex){
 const s=String(ex?.load||'');
 const num=s.match(/-?\d+(?:[.,]\d+)?/);
 return num?String(num[0].replace(',','.')):'';
}
function proMonthDate(s){
 const v=new Date(String(s||'')+'T12:00:00');
 return Number.isNaN(+v)?null:v;
}
function proWeekMonday(date){
 const d=proMonthDate(date)||new Date(),day=(d.getDay()+6)%7;
 d.setDate(d.getDate()-day);d.setHours(12,0,0,0);return d;
}
function proIso(d){
 return d.getFullYear()+'-'+String(d.getMonth()+1).padStart(2,'0')+'-'+String(d.getDate()).padStart(2,'0');
}
function proWeeklySessions(weekDate){
 const m=proWeekMonday(weekDate),start=proIso(m),endD=new Date(m);endD.setDate(endD.getDate()+7);const end=proIso(endD);
 return state.sessions.filter(s=>String(s.date)>=start&&String(s.date)<end);
}
function proWeeklyGoal(){
 const n=Number(state.settings.workoutWeeklyGoal);
 return n>=1&&n<=7?n:4;
}
function proWeekProgress(){
 const count=proWeeklySessions(inputDate()).length,goal=proWeeklyGoal();
 return {count,goal,percent:Math.min(100,Math.round(count/goal*100))};
}
function proSessionCountedByDay(){
 return new Set(proWeeklySessions(inputDate()).map(s=>s.date));
}

  const date=String(f.get('date')||r.date),duration=Math.max(1,Number(f.get('duration'))||1),rpe=Number(f.get('rpe'));

  workoutRunner=null;await saveWorkoutRunner();state.workoutPane='storico';toast('Allenamento registrato con le serie e i carichi reali.');

async function proCompleteSet(){
 const r=workoutRunner,x=runnerCurrent();if(!r||!x)return;
 const idx=r.exerciseIndex;
 const reps=Number(document.querySelector('#runnerActualReps')?.value);
 const loadRaw=String(document.querySelector('#runnerActualLoad')?.value||'').replace(',','.');
 const load=loadRaw===''?null:Number(loadRaw);
 if(!Number.isInteger(reps)||reps<0||reps>200){toast('Inserisci ripetizioni valide (0–200).');return;}
 if(load!==null&&(!Number.isFinite(load)||load<0||load>1000)){toast('Inserisci un carico valido.');return;}
 const previous=proSetDetails()[idx].length;
 if((r.completed[idx]||0)>=Math.max(1,Number(x.sets)||1))return;
 r.setDetails[idx].push({reps,load,at:Date.now()});
 r.lastSetRef={index:idx,offset:previous};
 try{await runnerSetDone();}catch(e){r.setDetails[idx].pop();toast('Errore di salvataggio: '+e.message);}
}
async function proUndoSet(){
 const r=workoutRunner,ref=r?.lastSetRef;if(!ref)return;
 if(!r.completed?.[ref.index]||!proSetDetails()[ref.index]?.length)return;
 r.completed[ref.index]--;
 r.setDetails[ref.index].pop();
 r.exerciseIndex=ref.index;
 r.lastSetRef=null;
 r.timer=freshTimer();
 await saveWorkoutRunner();render();toast('Ultima serie annullata.');
}
async function handleWorkoutProAction(act,t){
 switch(act){
  case 'proCompleteSet':await proCompleteSet();return true;
  case 'proUndo':await proUndoSet();return true;
  case 'proFinish':proSessionFinish();return true;
  case 'finishWorkout':proSessionFinish();return true;
  case 'proGoal':{
   const goal=proWeeklyGoal();
   openModal('<h2>Obiettivo settimanale</h2><p>Quante sessioni vuoi registrare in una settimana?</p><label class="field"><span>Allenamenti a settimana</span><select name="goal">'+[1,2,3,4,5,6,7].map(n=>'<option value="'+n+'" '+(goal===n?'selected':'')+'>'+n+'</option>').join('')+'</select></label>',async f=>{state.settings.workoutWeeklyGoal=Math.max(1,Math.min(7,Number(f.get('goal'))||4));await persist('settings');toast('Obiettivo aggiornato.');});
   return true;
  }
  case 'proExerciseInfo':proExerciseInfo(t.dataset.id,Number(t.dataset.index));return true;
  case 'proActiveInfo':if(runnerCurrent())openModal(proDetailHtml(runnerCurrent()),async()=>{},'Chiudi');return true;
  case 'proJump':{
   if(!workoutRunner)return true;
   const i=Number(t.dataset.index);
   if(!Number.isInteger(i)||i<0||i>=workoutRunner.exercises.length)return true;
   workoutRunner.exerciseIndex=i;await saveWorkoutRunner();render();return true;
  }
  case 'proAdjustReps':{
   const field=document.querySelector('#runnerActualReps');if(field)field.value=String(Math.max(0,Math.min(200,(Number(field.value)||0)+Number(t.dataset.n||0))));return true;
  }
  case 'proAdjustLoad':{
   const field=document.querySelector('#runnerActualLoad');if(field)field.value=String(Math.max(0,Math.min(1000,Math.round(((Number(field.value)||0)+Number(t.dataset.n||0))*2)/2)));return true;
  }
  case 'proSessionDetails':{
   const obj=state.sessions.find(s=>s.id===t.dataset.id);if(!obj)return true;
   const detail=Array.isArray(obj.setDetails)?obj.setDetails:[];
   const html=detail.map((sets,i)=>sets.length?'<div class="pro-summary-line"><strong>'+proEsc(obj.exercises?.[i]?.name||'Esercizio')+'</strong><span>'+sets.map((z,j)=>(j+1)+': '+z.reps+' × '+(z.load===null?'corpo libero':z.load+' kg')).join(' · ')+'</span></div>':'').join('');
   openModal('<h2>'+proEsc(obj.type||'Allenamento')+'</h2><p>'+proEsc(obj.date)+' · '+fmt(obj.duration)+' min · RPE '+proEsc(obj.rpe||'—')+'</p><div class="pro-summary-list">'+(html||'<p>Questa sessione è stata registrata prima del diario per serie.</p>')+'</div><p>'+proEsc(obj.notes||'')+'</p>',async()=>{},'Chiudi');
   return true;
  }
 }
 return false;
}
