function rf6DisplayDuration(mins){
 const m=Math.max(0,Math.round(Number(mins)||0));
 return m>=60?Math.floor(m/60)+'h '+String(m%60).padStart(2,'0')+'m':m+' min';
}
function rf6SessionPicture(session){
 const name=String(session?.type||session?.name||'');
 const t=(state.templates||[]).find(t=>String(t.name).toLowerCase()===name.toLowerCase()||String(t.kind).toLowerCase()===name.toLowerCase());
 const x=t?.exercises?.[0];
 return x?proExerciseIllustration(x,1,'rf6-recent-pose'):'<span class="rf6-recent-generic">✓</span>';
}
function rf6TodayShift(){
 const d=inputDate(),s=shiftFor(d),rec=recommended(d);
 const source=s?'Supershift':'Turno non disponibile';
 return '<div class="rf6-today-shift"><span class="rf6-activity-icon">▦</span><div><strong>'+(s?proEsc(s.label||s.type):'Oggi')+'</strong><small>'+(s?proEsc(shiftTime(s)):'Consulta la turnistica')+'</small></div><span class="rf6-shift-source">'+source+'</span></div>'+
 '<div class="rf6-suggestion"><strong>'+proEsc(rec.name)+'</strong><p>'+proEsc(rec.why)+'</p></div>'+
 '<button class="rf6-text-link" type="button" data-act="nav" data-to="turni">Vai alla turnistica →</button>';
}
renderToday=function(){

 const weekly=proWeeklySessions(inputDate());
 const duration=weekly.reduce((sum,x)=>sum+(Number(x.duration)||0),0);
 const volume=weekly.reduce((sum,x)=>sum+(typeof thirdVolumeBySession==='function'?thirdVolumeBySession(x):0),0);
 const records=(state.sessions||[]).slice().filter(s=>s?.date).sort((a,b)=>String(b.date).localeCompare(String(a.date))).slice(0,3);
 const health=(state.health?.workouts||[]).slice().filter(x=>x?.date).sort((a,b)=>String(b.date).localeCompare(String(a.date))).slice(0,3);
 const recent=records.length?records:health;
 const date=inputDate(),logs=mealDaily(date),sum=calcMacros(logs);

  '<div class="rf6-recent-row"><div class="rf6-recent-art">'+rf6SessionPicture(s)+'</div>'+

 ).join('');

 '</div>'+

 '<div class="rf6-dashboard-panels"><section class="rf6-info-panel"><div class="rf6-section-title"><h3>Il tuo turno</h3><span>Oggi</span></div>'+

// Keep existing tabs and handlers; change only text labels in the navigation for a native-app feel.
const rf6PreviousRenderNav=renderNav;
renderNav=function(){
 rf6PreviousRenderNav();
 const labels={oggi:'Home',allenamenti:'Allenati',alimentazione:'Diario',progressi:'Statistiche',turni:'Turni'};
 const nav=document.querySelector('#mobileNav');
 if(nav)nav.querySelectorAll('button[data-to]').forEach(button=>{
  const label=labels[button.dataset.to];
  if(!label)return;
  for(const node of button.childNodes)if(node.nodeType===3)node.textContent=label;
 });
};
