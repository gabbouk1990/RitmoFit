function defaultProgramMigration(templates){
 const existing=Array.isArray(templates)?templates:[],next=[...existing],matches=t=>JSON.stringify(t);
 let upgraded=0,protectedCount=0;
 for(const suggested of TEMPLATES){
  const pos=next.findIndex(t=>t.id===suggested.id);
  if(pos<0){next.push(structuredClone(suggested));upgraded++;continue;}
  const current=next[pos],legacy=LEGACY_TEMPLATES.find(t=>t.id===current.id);
  if(legacy&&matches(current)===matches(legacy)){next[pos]=structuredClone(suggested);upgraded++;}
  else if(matches(current)!==matches(suggested)) protectedCount++;
 }
 return {templates:next,upgraded,protectedCount};
}
async function migrateTrainingProgram(){
 const stored=await getItem('trainingProgramVersion',null);
 if(stored===TRAINING_PROGRAM_VERSION)return;
 const result=defaultProgramMigration(state.templates);
 state.templates=result.templates;
 if(result.upgraded)await persist('templates');
 await setItem('trainingProgramVersion',TRAINING_PROGRAM_VERSION);
}
async function restoreTrainingProgram(){
 if(!confirm('Caricare le schede aggiornate? Le versioni personalizzate delle schede di base verranno conservate come copie; storico, Salute, alimentazione e turni resteranno invariati.'))return;
 const ids=new Set(TEMPLATES.map(t=>t.id)),copies=[];
 for(const t of state.templates.filter(t=>ids.has(t.id))){
  const old=LEGACY_TEMPLATES.find(x=>x.id===t.id),isDefault=old&&JSON.stringify(t)===JSON.stringify(old);
  const isRecommended=JSON.stringify(t)===JSON.stringify(TEMPLATES.find(x=>x.id===t.id));
  if(!isDefault&&!isRecommended){copies.push({...structuredClone(t),id:'copia-'+t.id+'-'+Date.now()+'-'+copies.length,name:'Copia precedente · '+t.name});}
 }
 state.templates=[...TEMPLATES.map(t=>structuredClone(t)),...state.templates.filter(t=>!ids.has(t.id)),...copies];
 await persist('templates');await setItem('trainingProgramVersion',TRAINING_PROGRAM_VERSION);
 render();toast('Schede aggiornate. Le eventuali personalizzazioni precedenti sono state salvate come copie.');
}
