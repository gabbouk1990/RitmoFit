let rf5Draft=null;
function rf5Num(x,def=0){const n=Number(x);return Number.isFinite(n)?n:def;}
function rf5Current(){
 const t=rf4Program(rf4TemplateId);

}
function rf5Band(ex){return /elastic|band|resistance.?band/i.test(String(ex?.name||'')+' '+String(ex?.load||''));}
function rf5Model(ex){
 const text=String(ex?.reps||''),num=Number(text.match(/\d+/)?.[0])||10;
 const original=String(ex?.load||'');
 let color=/nero|black/i.test(original)?'nero':/ross|red/i.test(original)?'rosso':/giall|yellow/i.test(original)?'giallo':'nessuno';
 const bareWeight=Number(original.replace(',','.').match(/\d+(?:\.\d+)?/)?.[0]);
 return {
  sets:Math.max(1,Math.min(12,Math.round(rf5Num(ex?.sets,3)))),
  reps:Math.max(1,Math.min(proIsDistance(ex)?5000:200,num)),
  rest:Math.max(0,Math.min(600,restSeconds(ex?.rest)||60)),
  load:Number.isFinite(bareWeight)?bareWeight:0,
  band:ex?.bandColor||color,
  focus:ex?.focus||'standard',
  originalLoad:original
 };
}
function rf5DraftFor(){
 const {t,i,ex}=rf5Current();
 if(!t||!ex)return null;
 const key=t.id+'::'+i;
 if(!rf5Draft||rf5Draft.key!==key)rf5Draft={key,...rf5Model(ex),dirty:false};
 return rf5Draft;
}
function rf5Options(ex,field,values){

}

async function rf5SaveDraft(start=false){
 const {t,i,ex}=rf5Current(),draft=rf5DraftFor();
 if(!t||!ex||!draft)return;
 const original=structuredClone(ex);
 const newReps=proIsDistance(ex)?draft.reps+' m':String(draft.reps);
 const next={...ex,sets:draft.sets,reps:newReps,rest:draft.rest+' s',focus:draft.focus};
 if(rf5Band(ex)){
  next.bandColor=draft.band;
  next.load=draft.band==='nessuno'?'Corpo libero':'Elastico '+{giallo:'giallo leggero',rosso:'rosso medio',nero:'nero forte'}[draft.band];
 }else if(Number.isFinite(draft.load)){
  // Retain descriptive non-weight loads (e.g. bodyweight, technique, elastic).
  if(/kg|\d/.test(draft.originalLoad)||draft.load>0)next.load=draft.load>0?draft.load+' kg':'Corpo libero';
 }
 t.exercises[i]=next;
 try{await persist('templates');}
 catch(e){t.exercises[i]=original;toast('Impossibile salvare la scheda: '+e.message);return;}
 draft.dirty=false;
 if(start){await startWorkout(t.id);}
 else{toast('Esercizio aggiornato nella scheda.');render();}
}
// Extend existing delegated dispatch without replacing old timer, diary or shifts handlers.
const rf5OldHandle=handleWorkoutProAction;
handleWorkoutProAction=async function(act,t){
 if(act==='rf5Step'){
  const d=rf5DraftFor();if(!d)return true;
  const field=t.dataset.field,dir=Number(t.dataset.delta);
  const steps={sets:1,reps:proIsDistance(rf5Current().ex)?25:1,rest:15,load:2.5};
  if(!Object.hasOwn(steps,field)||![1,-1].includes(dir))return true;
  const low=field==='sets'||field==='reps'?1:0,max={sets:12,reps:proIsDistance(rf5Current().ex)?5000:200,rest:600,load:1000}[field];
  d[field]=Math.max(low,Math.min(max,Math.round((d[field]+steps[field]*dir)*10)/10));
  d.dirty=true;
  const box=document.querySelector('[data-rf5-value="'+field+'"]');
  if(box)box.textContent=d[field]+(field==='rest'?' s':field==='load'?' kg':field==='reps'&&proIsDistance(rf5Current().ex)?' m':'');
  return true;
 }
 if(act==='rf5Pick'){
  const d=rf5DraftFor(),field=t.dataset.field,value=t.dataset.value;if(!d)return true;
  if(field==='band'&&['nessuno','giallo','rosso','nero'].includes(value))d.band=value;
  else if(field==='focus'&&['standard','tecnica','forza','controllo'].includes(value))d.focus=value;
  else return true;
  d.dirty=true;
  const container=t.closest('.rf5-choice-row');
  container?.querySelectorAll('.rf5-chip').forEach(b=>{const on=b.dataset.value===value;b.classList.toggle('active',on);b.setAttribute('aria-pressed',String(on));});
  return true;
 }
 if(act==='rf5Save'||act==='rf5SaveStart'){await rf5SaveDraft(act==='rf5SaveStart');return true;}
 if(act==='rf4OpenExercise'||act==='rf4ExerciseNext'||act==='rf4ExercisePrev'){rf5Draft=null;}
 return rf5OldHandle(act,t);
};
