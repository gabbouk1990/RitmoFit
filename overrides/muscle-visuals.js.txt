function proMuscleKey(ex) {
 const n=String(ex?.name||ex||'').toLocaleLowerCase('it-IT').normalize('NFD').replace(/[\u0300-\u036f]/g,'');
 const matches=[
  [/nuoto|piscina|pull buoy|stile|dorso|defaticamento|scivolamento|drill|tecnica.*stile|aerobico|fondo/i,'swimming'],
  [/pallof|dead bug|antirotazion/i,'core'],
  [/suitcase march|farmer hold|farmer carry/i,'carry'],
  [/plank/i,'plank'],
  [/hip thrust|glute bridge|ponte glutei/i,'glutes'],
  [/stacco rumeno|(?:^|[^a-z])rdl(?:[^a-z]|$)|romanian deadlift|pull.through/i,'hinge'],
  [/goblet|squat|leg press|affondi/i,'squat'],
  [/panca inclinata|incline dumbbell/i,'incline'],
  [/panca|chest press|push.up|dip\b/i,'chest'],
  [/pullover|straight.arm pulldown/i,'pullover'],
  [/alzate laterali|lateral raise/i,'lateral'],
  [/shoulder press|military press|seated dumbbell press/i,'shoulders'],
  [/lat machine|lat pulldown|trazioni|pull.up/i,'vertical'],
  [/rematore|high row|chest.supported|machine row|seated row/i,'row'],
  [/skillmill|camminata|power walk|treadmill/i,'walking'],
  [/cyclette/i,'cycling']
 ];
 const found=matches.find(([expr])=>expr.test(n));
 return found?found[1]:null;
}

}
