const FOOD_CATALOG=FOOD_CATALOG_ROWS.map(([id,category,name,detail,portion,kcal,p,c,f])=>({id,category,name,detail,portion,kcal,p,c,f}));
const FOOD_BY_ID=new Map(FOOD_CATALOG.map(x=>[x.id,x]));
const FOOD_DEFAULT_RECIPES={
 colazione:[['yogurt-greco',170],['whey',15],['banana',120]],
 pranzo:[['pollo',120],['patate',220],['broccoli',200],['olio',10]],
 spuntino:[['whey',30],['banana',120]],
 cena:[['tonno',120],['fagioli',130],['pomodori',150],['pane-int',70],['olio',10]]
};
const nutritionKeys=['kcal','p','c','f'];
const nutritionRound=n=>Math.round((Number(n)||0)*10)/10;
function catalogIngredient(id,grams){
 const item=FOOD_BY_ID.get(id);if(!item)throw Error('Alimento non presente nel catalogo: '+id);
 return {foodId:item.id,category:item.category,name:item.name,detail:item.detail,grams:grams??item.portion,kcal:item.kcal,p:item.p,c:item.c,f:item.f};
}
function recipeIngredients(id){return (FOOD_DEFAULT_RECIPES[id]||[]).map(([key,g])=>catalogIngredient(key,g));}
function itemNutrition(row){
 const g=Math.max(0,Number(row.grams)||0)/100;
 return Object.fromEntries(nutritionKeys.map(k=>[k,nutritionRound(g*(Number(row[k])||0))]));
}
function foodNutrition(items){
 return items.reduce((sum,item)=>{const nutrition=itemNutrition(item);for(const k of nutritionKeys)sum[k]+=nutrition[k];return sum;},{kcal:0,p:0,c:0,f:0});
}
function upgradeDefaultMeals(){
 let changed=false;
 state.meals=state.meals.map(m=>{
  const legacy=MEALS.find(x=>x.id===m.id);
  if(!legacy||JSON.stringify(m)!==JSON.stringify(legacy)||!FOOD_DEFAULT_RECIPES[m.id])return m;
  const ingredients=recipeIngredients(m.id),nutrition=foodNutrition(ingredients);changed=true;
  return {...m,ingredients,...nutrition,details:m.details+' Quantità di esempio: personalizzabili.'};
 });
 return changed;
}
